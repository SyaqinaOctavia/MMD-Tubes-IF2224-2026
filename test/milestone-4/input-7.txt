program UjiKomprehensif;

var
  i, j, k: integer;
  arr : array [1..5] of array [1..5] of integer;

function fibonacci(n: integer): integer;
begin
  if (n <= 1) then
    fibonacci := n
  else
    fibonacci := fibonacci(n - 1) + fibonacci(n - 2);
end;

function faktorial(n: integer): integer;
begin
  if (n == 0) then
    faktorial := 1
  else
    faktorial := n * faktorial(n - 1);
end;
