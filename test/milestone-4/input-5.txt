program Hello;
var
  a : array [1..10] of array [1..10] of integer;
  b : integer;

function factorial(n: integer): integer;
begin
  if (n == 0) then
    factorial := 1
  else
    factorial := n * factorial(n - 1);  { <--- Panggilan rekursif }
end;
